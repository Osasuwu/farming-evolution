# Issue #88: the composter takes the mod's seeds, wild berries and products, by hand and from a
# hopper; a novice farmer trades bone meal; a working farmer leaves ripe plants of the mod alone.
# Every check prints a chat line with "say" only when it holds; "read" shows them.
# Run with gradlew runControl started: java tools/ClientControl.java --scenario tools/scenarios/farmer-compost.txt
command /time set 3000
command /gamerule doDaylightCycle false
command /clear @s
command /kill @e[type=item]
command /give @s farmingevolution:seeds[farmingevolution:genome={sex:"none",chromosomes:{species:{active:{id:"wild_flax"},inactive:{id:"wild_flax"}},fiber_strength:{active:7,inactive:4}}}] 16
command /give @s farmingevolution:wild_berries 4
command /give @s farmingevolution:wild_flax_fiber 4
# Composter by hand: seeds (slot 1), wild berries (slot 2), a product (slot 3), each into an empty composter.
tp -2.5 -60 -0.5 180 30
command /setblock -2 -60 -3 minecraft:composter
slot 1
lookat -2 -60 -3
use
command /execute if block -2 -60 -3 minecraft:composter[level=1] run say composter: seeds raised it to level 1
shot compost-seeds
command /setblock -2 -60 -3 minecraft:composter
slot 2
lookat -2 -60 -3
use
command /execute if block -2 -60 -3 minecraft:composter[level=1] run say composter: wild berries raised it to level 1
command /setblock -2 -60 -3 minecraft:composter
slot 3
lookat -2 -60 -3
use
command /execute if block -2 -60 -3 minecraft:composter[level=1] run say composter: wild flax fiber raised it to level 1
shot compost-fiber
# Composter fed by a hopper: 2 fibre and 2 seeds go in (too few to fill it) and the hopper ends empty.
command /setblock -4 -59 -3 minecraft:air
command /setblock -4 -60 -3 minecraft:composter
command /setblock -4 -59 -3 minecraft:hopper[facing=down]{Items:[{Slot:0b,id:"farmingevolution:wild_flax_fiber",count:2},{Slot:1b,id:"farmingevolution:seeds",count:2}]}
wait 200
command /execute unless block -4 -60 -3 minecraft:composter[level=0] if data block -4 -59 -3 {Items:[]} run say hopper: the composter took every item and is above level 0
shot compost-hopper
read 10
